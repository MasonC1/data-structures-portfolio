def recent_average(data, metric, season):
    previous = data[data["season"] < season]
    previous = previous.sort_values("season")
    
    return previous.tail(3)[metric].mean()

def linear_trend(data, metric, season):
    previous = data[data["season"] < season]

    X = previous[["season"]]
    y = previous[metric]

    model = LinearRegression()
    model.fit(X, y)

    prediction = model.predict([[season]])

    return prediction[0]

test_seasons = [2024, 2025]

forecast_results = []

for season in test_seasons:

    previous_data = love[love["season"] < season].copy()

    for metric in metrics:

        actual = love[
            love["season"] == season
        ][metric].iloc[0]

        # Career average
        career_prediction = previous_data[metric].mean()

        # Recent average
        recent_data = previous_data.sort_values("season").tail(2)
        recent_prediction = recent_data[metric].mean()

        # Linear trend
        X = previous_data[["season"]]
        y = previous_data[metric]

        model = LinearRegression()
        model.fit(X, y)

        trend_prediction = model.predict([[season]])[0]

        forecast_results.append({
            "season": season,
            "metric": metric,
            "actual": actual,
            "career_average": career_prediction,
            "recent_average": recent_prediction,
            "linear_trend": trend_prediction
        })
